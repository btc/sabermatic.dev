# Billing & Entitlements Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement a grant-based minute ledger with Stripe integration, entitlement enforcement on session creation, and feature gating for educator/coach access.

**Architecture:** Grant buckets track minute balances from various sources (free monthly, subscription, purchase). An append-only ledger records all mutations. Stripe handles money (Checkout, Portal, webhooks); we handle entitlements. Sessions reserve minutes upfront and refund unused on completion.

**Tech Stack:** Go, PostgreSQL (sqlc + pgx/v5), Stripe Go SDK (`github.com/stripe/stripe-go/v82`), existing Backend DI pattern.

**Base branch:** `ui-phase` (nearly code-complete, contains all phases 1-8 + UI)

**Spec:** `docs/superpowers/specs/2026-04-03-billing-design.md`

-- 004_billing.up.sql

-- Grant buckets: each source of minutes creates a grant.
CREATE TABLE grants (
    id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id           UUID NOT NULL REFERENCES users(id),
    source            TEXT NOT NULL
                      CHECK (source IN ('free_grant', 'subscription', 'purchase', 'admin')),
    stripe_event_id   TEXT UNIQUE,
    initial_minutes   INT NOT NULL CHECK (initial_minutes > 0),
    remaining_minutes INT NOT NULL CHECK (remaining_minutes >= 0),
    expires_at        TIMESTAMPTZ,
    created_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CHECK (remaining_minutes <= initial_minutes)
);

CREATE INDEX idx_grants_user_balance ON grants(user_id)
    WHERE remaining_minutes > 0;

CREATE UNIQUE INDEX idx_grants_free_per_month
    ON grants(user_id, date_trunc('month', created_at))
    WHERE source = 'free_grant';

CREATE INDEX idx_users_stripe_customer ON users(stripe_customer_id)
    WHERE stripe_customer_id IS NOT NULL;

-- Append-only audit log of all balance mutations.
CREATE TABLE ledger_entries (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id     UUID NOT NULL REFERENCES users(id),
    grant_id    UUID NOT NULL REFERENCES grants(id),
    amount      INT NOT NULL,
    reason      TEXT NOT NULL
                CHECK (reason IN (
                    'free_monthly', 'subscription_renewal', 'purchase', 'admin_grant',
                    'session_reserve', 'session_refund', 'error_refund'
                )),
    session_id  UUID REFERENCES interview_sessions(id),
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_ledger_user_created ON ledger_entries(user_id, created_at);
CREATE INDEX idx_ledger_session ON ledger_entries(session_id)
    WHERE session_id IS NOT NULL;

-- Track reserved minutes per session for refund calculation.
ALTER TABLE interview_sessions ADD COLUMN reserved_minutes INT;

-- Add 'failed' status for platform-error sessions that get full refunds.
ALTER TABLE interview_sessions DROP CONSTRAINT interview_sessions_status_check;
ALTER TABLE interview_sessions ADD CONSTRAINT interview_sessions_status_check
    CHECK (status IN ('active', 'completed', 'evaluating', 'reviewed', 'evaluation_failed', 'failed'));

-- Track free educator analysis taste-test usage (lifetime counter).
ALTER TABLE users ADD COLUMN free_full_educators_used INT NOT NULL DEFAULT 0;

-- usage_periods is superseded by grants + ledger.
DROP TABLE usage_periods;
```

-- name: GetUserBalance :one
SELECT COALESCE(SUM(remaining_minutes), 0)::int AS balance
FROM grants
WHERE user_id = $1
  AND remaining_minutes > 0
  AND (expires_at IS NULL OR expires_at > NOW());

-- name: GetUserPaidBalance :one
SELECT COALESCE(SUM(remaining_minutes), 0)::int AS balance
FROM grants
WHERE user_id = $1
  AND remaining_minutes > 0
  AND source != 'free_grant'
  AND (expires_at IS NULL OR expires_at > NOW());

-- name: SelectGrantsForReservation :many
SELECT id, remaining_minutes
FROM grants
WHERE user_id = $1
  AND remaining_minutes > 0
  AND (expires_at IS NULL OR expires_at > NOW())
ORDER BY expires_at ASC NULLS LAST
FOR UPDATE;

-- name: DebitGrant :one
UPDATE grants
SET remaining_minutes = remaining_minutes - $2
WHERE id = $1
  AND remaining_minutes >= $2
RETURNING remaining_minutes;

-- name: CreditGrant :one
UPDATE grants
SET remaining_minutes = remaining_minutes + $2
WHERE id = $1
  AND remaining_minutes + $2 <= initial_minutes
RETURNING remaining_minutes;

-- name: InsertLedgerEntry :one
INSERT INTO ledger_entries (user_id, grant_id, amount, reason, session_id)
VALUES ($1, $2, $3, $4, $5)
RETURNING *;

-- name: CreateGrantFromStripe :one
INSERT INTO grants (user_id, source, stripe_event_id, initial_minutes, remaining_minutes, expires_at)
VALUES ($1, $2, $3, $4, $4, $5)
ON CONFLICT (stripe_event_id) DO NOTHING
RETURNING *;

-- name: CreateFreeGrant :one
INSERT INTO grants (user_id, source, initial_minutes, remaining_minutes, expires_at)
VALUES ($1, 'free_grant', $2, $2, $3)
ON CONFLICT (user_id, date_trunc('month', created_at)) WHERE source = 'free_grant'
DO NOTHING
RETURNING *;

-- name: GetFreeGrantForMonth :one
SELECT id FROM grants
WHERE user_id = $1
  AND source = 'free_grant'
  AND expires_at = $2
LIMIT 1;

-- name: ListActiveGrants :many
SELECT id, source, initial_minutes, remaining_minutes, expires_at, created_at
FROM grants
WHERE user_id = $1
  AND remaining_minutes > 0
  AND (expires_at IS NULL OR expires_at > NOW())
ORDER BY expires_at ASC NULLS LAST;

-- name: GetRecentLedgerEntries :many
SELECT amount, reason, session_id, created_at
FROM ledger_entries
WHERE user_id = $1
ORDER BY created_at DESC
LIMIT $2;

-- name: GetSessionReservationEntries :many
SELECT grant_id, amount
FROM ledger_entries
WHERE session_id = $1 AND reason = 'session_reserve'
ORDER BY created_at DESC;

-- name: GetUserUsageSummary :one
SELECT
  COALESCE(SUM(remaining_minutes) FILTER (WHERE expires_at IS NULL OR expires_at > NOW()), 0)::int AS total_balance,
  COALESCE(SUM(remaining_minutes) FILTER (WHERE source = 'free_grant' AND (expires_at IS NULL OR expires_at > NOW())), 0)::int AS free_balance,
  COALESCE(SUM(remaining_minutes) FILTER (WHERE source != 'free_grant' AND (expires_at IS NULL OR expires_at > NOW())), 0)::int AS paid_balance
FROM grants
WHERE user_id = $1
  AND remaining_minutes > 0;

```

package billing

import "time"

// EducatorAccessLevel controls what educator content a user sees.
type EducatorAccessLevel int

const (
	Preview   EducatorAccessLevel = iota // Teaser summary only
	Full                                 // Complete deep-dive analysis
	FreeTaste                            // One-time full analysis for free users
)

// Plan defines the limits and features for a billing tier.
type Plan struct {
	Name               string
	MinutesPerMonth    int
	MaxDurationMinutes int
	ConcurrentSessions int
	CoachAccess        bool
	EducatorAccess     EducatorAccessLevel
	FreeEducatorLimit  int
	StripePriceID      string
}

// plans is the canonical plan registry. Map key is the programmatic identifier
// stored in users.plan. Name is the display label.
// Effectively immutable after init — do not modify at runtime.
var plans = map[string]Plan{
	"free": {
		Name:               "Free",
		MinutesPerMonth:    60,
		MaxDurationMinutes: 30,
		ConcurrentSessions: 1,
		CoachAccess:        false,
		EducatorAccess:     Preview,
		FreeEducatorLimit:  1,
	},
	"pro": {
		Name:               "Pro",
		MinutesPerMonth:    600,
		MaxDurationMinutes: 180,
		ConcurrentSessions: 3,
		CoachAccess:        true,
		EducatorAccess:     Full,
		FreeEducatorLimit:  0,
		StripePriceID:      "", // set via config at startup
	},
}

// PlanByName returns the plan for the given key and whether it exists.
func PlanByName(name string) (Plan, bool) {
	p, ok := plans[name]
	return p, ok
}

// SetProPriceID sets the Stripe price ID for the pro plan. Called once at startup.
func SetProPriceID(priceID string) {
	p := plans["pro"]
	p.StripePriceID = priceID
	plans["pro"] = p
}

// MinutePack represents a purchasable bundle of minutes.
type MinutePack struct {
	Minutes       int
	StripePriceID string
}

var packs = map[int]MinutePack{
	120: {Minutes: 120},
	300: {Minutes: 300},
	600: {Minutes: 600},
}

// PackByMinutes returns the pack for the given minute count and whether it exists.
func PackByMinutes(minutes int) (MinutePack, bool) {
	p, ok := packs[minutes]
	return p, ok
}

// SetPackPriceIDs sets Stripe price IDs for minute packs. Called once at startup.
func SetPackPriceIDs(pack120, pack300, pack600 string) {
	set := func(mins int, priceID string) {
		p := packs[mins]
		p.StripePriceID = priceID
		packs[mins] = p
	}
	set(120, pack120)
	set(300, pack300)
	set(600, pack600)
}

// AllPacks returns all available minute packs (for API responses).
func AllPacks() []MinutePack {
	return []MinutePack{packs[120], packs[300], packs[600]}
}

// EndOfMonth returns the first instant of the next month (used as free grant expiry).
func EndOfMonth(t time.Time) time.Time {
	y, m, _ := t.Date()
	return time.Date(y, m+1, 1, 0, 0, 0, 0, time.UTC)
}

package billing

import "context"

// DetermineEducatorAccess decides what level of educator content to show.
// paidBalance: user's non-free grant balance. freeUsed: user's free_full_educators_used.
// freeLimit: plan's FreeEducatorLimit.
func DetermineEducatorAccess(_ context.Context, paidBalance int, freeUsed int, freeLimit int) (EducatorAccessLevel, error) {
	if paidBalance > 0 {
		return Full, nil
	}
	if freeUsed < freeLimit {
		return FreeTaste, nil
	}
	return Preview, nil
}

// CanAccessCoach returns true if the user has paid minute balance.
func CanAccessCoach(paidBalance int) bool {
	return paidBalance > 0
}

// EnsureFreeGrant creates the current month's free grant if it doesn't exist.
// Race-safe via ON CONFLICT on the partial unique index.
func (b *Backend) EnsureFreeGrant(ctx context.Context, userID uuid.UUID) error {
	plan, ok := billing.PlanByName("free")
	if !ok || plan.MinutesPerMonth == 0 {
		return nil
	}
	expiresAt := billing.EndOfMonth(time.Now())
	_, err := db.New(b.pool).CreateFreeGrant(ctx, db.CreateFreeGrantParams{
		UserID:         userID,
		InitialMinutes: int32(plan.MinutesPerMonth),
		ExpiresAt:      pgtype.Timestamptz{Time: expiresAt, Valid: true},
	})
	if err != nil {
		// ON CONFLICT DO NOTHING returns no rows → pgx.ErrNoRows means grant already exists.
		if errors.Is(err, pgx.ErrNoRows) {
			return nil
		}
		return fmt.Errorf("ensure free grant: %w", err)
	}
	return nil
}

// ReserveMinutes atomically reserves minutes across grants using FIFO-by-expiry.
// Returns the total minutes reserved. Writes ledger entries for each grant touched.
func (b *Backend) ReserveMinutes(ctx context.Context, userID uuid.UUID, minutes int, sessionID uuid.UUID) (int, error) {
	tx, err := b.pool.Begin(ctx)
	if err != nil {
		return 0, fmt.Errorf("begin reserve tx: %w", err)
	}
	defer tx.Rollback(ctx) //nolint:errcheck

	q := db.New(tx)

	// Lock grants in FIFO order
	grants, err := q.SelectGrantsForReservation(ctx, userID)
	if err != nil {
		return 0, fmt.Errorf("select grants: %w", err)
	}

	// Check total available
	var total int
	for _, g := range grants {
		total += int(g.RemainingMinutes)
	}
	if total < minutes {
		return 0, ErrInsufficientBalance
	}

	// Walk grants, debit until reservation fulfilled
	remaining := minutes
	for _, g := range grants {
		if remaining <= 0 {
			break
		}
		debit := remaining
		if debit > int(g.RemainingMinutes) {
			debit = int(g.RemainingMinutes)
		}

		_, err := q.DebitGrant(ctx, db.DebitGrantParams{
			ID:               g.ID,
			RemainingMinutes: int32(debit),
		})
		if err != nil {
			return 0, fmt.Errorf("debit grant %s: %w", g.ID, err)
		}

		_, err = q.InsertLedgerEntry(ctx, db.InsertLedgerEntryParams{
			UserID:    userID,
			GrantID:   g.ID,
			Amount:    int32(-debit),
			Reason:    "session_reserve",
			SessionID: pgtype.UUID{Bytes: sessionID, Valid: true},
		})
		if err != nil {
			return 0, fmt.Errorf("insert ledger entry: %w", err)
		}

		remaining -= debit
	}

	if err := tx.Commit(ctx); err != nil {
		return 0, fmt.Errorf("commit reserve: %w", err)
	}

	return minutes, nil
}

// RefundMinutes credits minutes back to the grants that were originally debited.
// reason should be "session_refund" or "error_refund".
func (b *Backend) RefundMinutes(ctx context.Context, userID uuid.UUID, sessionID uuid.UUID, minutes int, reason string) error {
	tx, err := b.pool.Begin(ctx)
	if err != nil {
		return fmt.Errorf("begin refund tx: %w", err)
	}
	defer tx.Rollback(ctx) //nolint:errcheck

	q := db.New(tx)

	// Reconstruct per-grant reservation amounts from ledger
	entries, err := q.GetSessionReservationEntries(ctx, sessionID)
	if err != nil {
		return fmt.Errorf("get reservation entries: %w", err)
	}

	// Walk in reverse (newest-debited first), credit back
	remaining := minutes
	for _, e := range entries {
		if remaining <= 0 {
			break
		}
		originalDebit := int(-e.Amount) // amounts are negative for debits
		credit := remaining
		if credit > originalDebit {
			credit = originalDebit
		}

		_, err := q.CreditGrant(ctx, db.CreditGrantParams{
			ID:               e.GrantID,
			RemainingMinutes: int32(credit),
		})
		if err != nil {
			slog.Warn("credit grant failed (may be expired)", "grant_id", e.GrantID, "error", err)
			// Grant may have been expired — the credit is lost. Continue to next grant.
			continue
		}

		_, err = q.InsertLedgerEntry(ctx, db.InsertLedgerEntryParams{
			UserID:    userID,
			GrantID:   e.GrantID,
			Amount:    int32(credit),
			Reason:    reason,
			SessionID: pgtype.UUID{Bytes: sessionID, Valid: true},
		})
		if err != nil {
			return fmt.Errorf("insert refund ledger entry: %w", err)
		}

		remaining -= credit
	}

	if err := tx.Commit(ctx); err != nil {
		return fmt.Errorf("commit refund: %w", err)
	}

	return nil
}

func (b *Backend) CreateSession(ctx context.Context, p CreateSessionParams) (db.InterviewSession, error) {
	if p.DurationMinutes < 1 || p.DurationMinutes > 180 {
		return db.InterviewSession{}, ErrInvalidDuration
	}

	// Look up plan config and validate duration against plan max.
	plan, ok := billing.PlanByName(p.Plan)
	if !ok {
		plan, _ = billing.PlanByName("free")
	}
	if p.DurationMinutes > plan.MaxDurationMinutes {
		return db.InterviewSession{}, ErrDurationExceedsPlan
	}

	// Ensure current-month free grant exists for ALL users (including pro).
	// Pro users also get free monthly minutes per spec. Race-safe via ON CONFLICT.
	if err := b.EnsureFreeGrant(ctx, p.UserID); err != nil {
		return db.InterviewSession{}, err
	}

	// Begin transaction for authoritative enforcement.
	tx, err := b.pool.Begin(ctx)
	if err != nil {
		return db.InterviewSession{}, fmt.Errorf("begin session tx: %w", err)
	}
	defer tx.Rollback(ctx) //nolint:errcheck

	q := db.New(tx)

	// Check concurrent sessions.
	activeCount, err := q.CountActiveSessionsByUser(ctx, p.UserID)
	if err != nil {
		return db.InterviewSession{}, fmt.Errorf("count active sessions: %w", err)
	}
	if int(activeCount) >= plan.ConcurrentSessions {
		return db.InterviewSession{}, ErrConcurrentSessionLimit
	}

	// Verify question exists.
	if _, err := q.GetQuestion(ctx, p.QuestionID); err != nil {
		if errors.Is(err, pgx.ErrNoRows) {
			return db.InterviewSession{}, ErrQuestionNotFound
		}
		return db.InterviewSession{}, fmt.Errorf("get question: %w", err)
	}

	// Lock grants and check balance.
	grants, err := q.SelectGrantsForReservation(ctx, p.UserID)
	if err != nil {
		return db.InterviewSession{}, fmt.Errorf("select grants: %w", err)
	}
	var totalAvailable int
	for _, g := range grants {
		totalAvailable += int(g.RemainingMinutes)
	}
	if totalAvailable < p.DurationMinutes {
		return db.InterviewSession{}, ErrInsufficientBalance
	}

	// Create session first to get the session ID.
	session, err := q.CreateSession(ctx, db.CreateSessionParams{
		UserID:                p.UserID,
		QuestionID:            p.QuestionID,
		ConfigDurationMinutes: int32(p.DurationMinutes),
		ConfigTtsEnabled:      p.TTSEnabled,
	})
	if err != nil {
		return db.InterviewSession{}, fmt.Errorf("create session: %w", err)
	}

	// FIFO debit across grants.
	remaining := p.DurationMinutes
	for _, g := range grants {
		if remaining <= 0 {
			break
		}
		debit := remaining
		if debit > int(g.RemainingMinutes) {
			debit = int(g.RemainingMinutes)
		}
		if _, err := q.DebitGrant(ctx, db.DebitGrantParams{
			ID:               g.ID,
			RemainingMinutes: int32(debit),
		}); err != nil {
			return db.InterviewSession{}, fmt.Errorf("debit grant: %w", err)
		}
		if _, err := q.InsertLedgerEntry(ctx, db.InsertLedgerEntryParams{
			UserID:    p.UserID,
			GrantID:   g.ID,
			Amount:    int32(-debit),
			Reason:    "session_reserve",
			SessionID: pgtype.UUID{Bytes: session.ID, Valid: true},
		}); err != nil {
			return db.InterviewSession{}, fmt.Errorf("insert ledger: %w", err)
		}
		remaining -= debit
	}

	// Set reserved_minutes on the session.
	if err := q.UpdateSessionReservedMinutes(ctx, db.UpdateSessionReservedMinutesParams{
		ID:              session.ID,
		ReservedMinutes: pgtype.Int4{Int32: int32(p.DurationMinutes), Valid: true},
	}); err != nil {
		return db.InterviewSession{}, fmt.Errorf("set reserved minutes: %w", err)
	}

	if err := tx.Commit(ctx); err != nil {
		return db.InterviewSession{}, fmt.Errorf("commit session: %w", err)
	}

	return session, nil
}

	if err != nil {

	if session.ReservedMinutes.Valid && session.ReservedMinutes.Int32 > 0 {
		actualMinutes := 1 // minimum 1 minute
		if session.EndedAt.Valid {
			// StartedAt is time.Time (NOT NULL), EndedAt is pgtype.Timestamptz (nullable).

// CreateCheckoutSession creates a Stripe Checkout session for subscription or pack purchase.
// Looks up user's stripe_customer_id from DB; creates a Stripe Customer if none exists.
func (b *Backend) CreateCheckoutSession(ctx context.Context, userID uuid.UUID, email string, checkoutType string, planName string, packMinutes int) (string, error) {
	if b.cfg.Stripe.SecretKey == "" {
		return "", fmt.Errorf("stripe not configured")
	}

	// Look up user's stripe_customer_id from DB.
	q := db.New(b.pool)
	user, err := q.GetUserByID(ctx, userID)
	if err != nil {
		return "", fmt.Errorf("get user: %w", err)
	}
	custID := user.StripeCustomerID.String
	if !user.StripeCustomerID.Valid || custID == "" {
		params := &stripe.CustomerParams{
			Email: stripe.String(email),
			Params: stripe.Params{
				Metadata: map[string]string{"user_id": userID.String()},
			},
		}
		c, err := customer.New(params)
		if err != nil {
			return "", fmt.Errorf("create stripe customer: %w", err)
		}
		custID = c.ID
		// Persist the customer ID.
		db.New(b.pool).UpdateUserStripeCustomerID(ctx, db.UpdateUserStripeCustomerIDParams{
			ID:               userID,
			StripeCustomerID: pgtype.Text{String: custID, Valid: true},
		})
	}

	var mode stripe.CheckoutSessionMode
	var priceID string

	switch checkoutType {
	case "subscription":
		mode = stripe.CheckoutSessionModeSubscription
		plan, ok := billing.PlanByName(planName)
		if !ok || plan.StripePriceID == "" {
			return "", fmt.Errorf("invalid plan: %s", planName)
		}
		priceID = plan.StripePriceID
	case "pack":
		mode = stripe.CheckoutSessionModePayment
		pack, ok := billing.PackByMinutes(packMinutes)
		if !ok || pack.StripePriceID == "" {
			return "", fmt.Errorf("invalid pack: %d minutes", packMinutes)
		}
		priceID = pack.StripePriceID
	default:
		return "", fmt.Errorf("invalid checkout type: %s", checkoutType)
	}

	params := &stripe.CheckoutSessionParams{
		Customer: stripe.String(custID),
		Mode:     stripe.String(string(mode)),

import (
	"io"
	"github.com/stripe/stripe-go/v82/webhook"

// PostStripeWebhook handles Stripe webhook events.
func PostStripeWebhook(b *backend.Backend) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		body, err := io.ReadAll(io.LimitReader(r.Body, 65536))
		if err != nil {
			http.Error(w, "read body", http.StatusBadRequest)
			return
		}

		sig := r.Header.Get("Stripe-Signature")
		event, err := webhook.ConstructEvent(body, sig, b.Config().Stripe.WebhookSecret)
		if err != nil {
			http.Error(w, "invalid signature", http.StatusBadRequest)
			return
		}

		if err := b.HandleStripeWebhook(r.Context(), event); err != nil {
			// Log but return 200 to prevent infinite retries for bugs.
			slog.Error("stripe webhook handler error", "type", event.Type, "error", err)
		}

		w.WriteHeader(http.StatusOK)
	}
}

// FailSession transitions a session to "failed" and refunds all reserved minutes.
func (b *Backend) FailSession(ctx context.Context, sessionID uuid.UUID) error {
	tx, err := b.pool.Begin(ctx)
	if err != nil {
		return fmt.Errorf("begin fail-session tx: %w", err)
	}
	defer tx.Rollback(ctx) //nolint:errcheck

	q := db.New(tx)
	session, err := q.GetSessionByID(ctx, sessionID)
	if err != nil {
		return fmt.Errorf("get session: %w", err)
	}
	if session.Status != "active" {
		return ErrSessionNotActive
	}

	// Use UpdateSessionStatusOnly (not UpdateSessionStatus) to avoid
	// overwriting turn_count or ended_at on a partially-completed session.
	if err := q.UpdateSessionStatusOnly(ctx, db.UpdateSessionStatusOnlyParams{
		ID:     sessionID,
		Status: "failed",
	}); err != nil {
		return fmt.Errorf("update session status: %w", err)
	}

	// Full refund of reserved minutes.
	if session.ReservedMinutes.Valid && session.ReservedMinutes.Int32 > 0 {
		if err := b.refundMinutesTx(ctx, tx, session.UserID, sessionID, int(session.ReservedMinutes.Int32), "error_refund"); err != nil {
			slog.Warn("fail-session refund failed", "session_id", sessionID, "error", err)
		}
	}

	return tx.Commit(ctx)
}

async function handleCheckout(type: "subscription" | "pack", plan?: string, minutes?: number) {
  setCheckoutLoading(true);
  setCheckoutError(null);
  try {
    const body = type === "subscription"
      ? { type: "subscription", plan: plan ?? "pro" }
      : { type: "pack", minutes };
    const res = await apiClient.post<{ url: string }>("/api/billing/checkout", body);
    if (res?.url) {
      window.location.href = res.url;
    }
  } catch (err) {
    const msg = err instanceof ApiError
      ? `Could not start checkout (${err.status}). Please try again.`
      : "Could not start checkout. Please try again.";
    setCheckoutError(msg);
    setCheckoutLoading(false);
  }
}
